#   uv venv --python 3.11 .venv-pocket

#   uv pip install --python .venv-pocket -e .            # el core: pydantic + pyyaml

# aborta al cargar, pero si alguna vez se saca ese chequeo, el sintoma es una similitud baja que
# se lee como un modelo malo.
pocket-tts==4.0.1

# PIN OBLIGATORIO por el driver de esta maquina. `pocket-tts` declara `torch>=2.5.0` sin tope, asi
# que la resolucion trae el wheel mas nuevo del indice; si su CUDA es mas nueva que el driver,
# `torch.cuda.is_available()` queda en False con un UserWarning y la corrida de GPU mide CPU.
# 2.6.0+cu126 es el mismo que usa envs/chatterbox_v3.txt.
